    // NANO BANANA 2.1 (gemini-nano-banana-2.1) API CALLER

    // =========================================================================

    async function callNanoBanana(promptText, base64ImagesArray, stepNumber) {

      const apiKey = document.getElementById('apiKeyInput').value.trim();

      const selectedModel = document.getElementById('modelSelect').value;

      const thinkingLevel = document.getElementById('thinkingLevelSelect').value;

      const imageSize = document.getElementById('imageSizeSelect').value;


      // Run visual simulation if no API key is provided or Demo mode is chosen

      if (!apiKey || selectedModel === 'demo') {

        return await simulateNanoBananaStep(stepNumber);

      }


      const parts = [{ text: promptText }];

      for (const dataUrl of base64ImagesArray) {

        const [meta, base64Data] = dataUrl.split(',');

        const mimeType = meta.match(/:(.*?);/)[1];

        parts.push({

          inline_data: {

            mime_type: mimeType,

            data: base64Data

          }

        });

      }


      // Official Gemini Nano Banana 2.1 Endpoint: gemini-nano-banana-2.1

      const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${selectedModel}:generateContent?key=${apiKey}`;

      const payload = {

        contents: [{ parts }],

        generationConfig: {

          responseModalities: ["TEXT", "IMAGE"],

          maxOutputTokens: 32768,

          thinkingConfig: {

            thinkingLevel: thinkingLevel

          },

          imageConfig: {

            imageSize: imageSize

          }

        }

      };


      const response = await fetch(endpoint, {

        method: 'POST',

        headers: { 'Content-Type': 'application/json' },

        body: JSON.stringify(payload)

      });


      if (!response.ok) {

        const errText = await response.text();

        throw new Error(`Nano Banana 2.1 API Error (${response.status}): ${errText}`);

      }


      const result = await response.json();

      const candidateParts = result.candidates?.[0]?.content?.parts || [];

      for (const part of candidateParts) {

        const inlineData = part.inlineData || part.inline_data;

        if (inlineData && inlineData.data) {

          const mime = inlineData.mimeType || inlineData.mime_type || 'image/png';

          return `data:${mime};base64,${inlineData.data}`;

        }

      }


      throw new Error("No image was returned by gemini-nano-banana-2.1. Verify your API key and quota.");

    }

    async function simulateNanoBananaStep(stepNumber) {

      await new Promise(r => setTimeout(r, 1100));

      const offCanvas = document.createElement('canvas');

      offCanvas.width = 960;

      offCanvas.height = 540;

      const c = offCanvas.getContext('2d');


      const prevImg = new Image();

      const sourceUrl = stepNumber === 1 ? chainState.images[0] : (chainState.images[stepNumber - 1] || chainState.images[0]);

      await new Promise(resolve => {

        prevImg.onload = resolve;

        prevImg.src = sourceUrl;

      });

      c.drawImage(prevImg, 0, 0, 960, 540);


      if (stepNumber === 1) {

        c.fillStyle = '#cbd5e1';

        c.fillRect(0, 0, 960, 380);

        c.fillStyle = '#64748b';

        c.fillRect(0, 380, 960, 160);


        c.fillStyle = '#94a3b8';

        c.fillRect(160, 290, 640, 100);

        c.fillStyle = '#0f172a';

        for (let x = 190; x < 760; x += 95) {

          c.fillRect(x, 310, 60, 80);

        }


        c.fillStyle = '#b45309';

        c.fillRect(190, 150, 340, 140);

        c.strokeStyle = '#78350f';

        for (let x = 195; x < 530; x += 8) {

          c.beginPath(); c.moveTo(x, 150); c.lineTo(x, 290); c.stroke();

        }


        c.fillStyle = '#38bdf8';

        c.globalAlpha = 0.45;

        c.fillRect(530, 170, 240, 120);

        c.globalAlpha = 1.0;

        c.strokeStyle = '#1e293b';

        for (let x = 530; x <= 770; x += 30) {

          c.beginPath(); c.moveTo(x, 170); c.lineTo(x, 290); c.stroke();

        }


        c.fillStyle = '#f8fafc';

        c.beginPath();

        c.moveTo(120, 150); c.lineTo(480, 95); c.lineTo(840, 145);

        c.lineTo(820, 170); c.lineTo(480, 125); c.lineTo(140, 170);

        c.closePath();

        c.fill();

      } else if (stepNumber === 2) {

        const grad = c.createLinearGradient(0, 0, 0, 540);

        grad.addColorStop(0, 'rgba(49, 46, 129, 0.45)');

        grad.addColorStop(0.5, 'rgba(217, 119, 6, 0.35)');

        grad.addColorStop(1, 'rgba(24, 24, 27, 0.45)');

        c.fillStyle = grad;

        c.fillRect(0, 0, 960, 540);

      } else if (stepNumber === 3) {

        c.fillStyle = '#14532d';

        c.beginPath(); c.arc(95, 280, 55, 0, Math.PI * 2); c.fill();

        c.beginPath(); c.arc(870, 270, 65, 0, Math.PI * 2); c.fill();

        c.fillStyle = '#3f2e21';

        c.fillRect(90, 320, 10, 80);

        c.fillRect(865, 320, 12, 80);


        c.fillStyle = '#090d16';

        const peopleCoords = [260, 310, 450, 580, 620];

        peopleCoords.forEach(px => {

          c.beginPath(); c.arc(px, 365, 5, 0, Math.PI * 2); c.fill();

          c.fillRect(px - 4, 372, 8, 24);

        });

      }


      return offCanvas.toDataURL('image/png');

    }
